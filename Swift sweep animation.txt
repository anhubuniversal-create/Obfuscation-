-- === 1. INVISIBILITY HELPERS ===
local function DisconnectAll()
	for _, conn in ipairs(Connections) do
		conn:Disconnect()
	end
	table.clear(Connections)
end

local function HideObject(object)
	if object:IsA("BasePart") then
		object.LocalTransparencyModifier = 1
	elseif object:IsA("Decal") or object:IsA("Texture") then
		object.Transparency = 1
	elseif object:IsA("ParticleEmitter")
		or object:IsA("Trail")
		or object:IsA("Beam")
		or object:IsA("Smoke")
		or object:IsA("Fire")
		or object:IsA("Sparkles") then
		object.Enabled = false
	end
end

local function HideCharacter(char)
	for _, object in ipairs(char:GetDescendants()) do
		HideObject(object)
	end
end

local function MakePart(name, size)
	local part = Instance.new("Part")
	part.Name = name
	part.Size = size
	part.Color = Color3.fromRGB(130, 130, 130)
	part.Material = Enum.Material.SmoothPlastic
	part.Anchored = true
	part.CanCollide = false
	part.CanTouch = false
	part.CanQuery = false
	part.Massless = true
	part.Parent = Proxy
	return part
end

local targetFPS = 55        
local playbackSpeed = 1     

local totalFrames = #frames
local totalDuration = ((totalFrames - 1) / targetFPS) / playbackSpeed
local origin = frames[1]:Inverse()

local function calculateMainTimeline(elapsedTime)
	if elapsedTime >= totalDuration then
		return totalFrames, 1, true
	end
	
	local progress = elapsedTime / totalDuration
	local scaledIndex = progress * (totalFrames - 1) + 1
	local index = math.floor(scaledIndex)
	local alpha = scaledIndex - index
	
	return index, alpha, false
end

	if invisibleBlock and invisibleBlock.Parent then
		invisibleBlock.CFrame = Hrp1.CFrame

		ProxyTorso.CFrame = Hrp0.CFrame
	end

	if Head then ProxyHead.CFrame = Hrp0.CFrame * (Hrp1.CFrame:Inverse() * Head.CFrame) end
	if LeftArm and ProxyLeftArm then ProxyLeftArm.CFrame = Hrp0.CFrame * (Hrp1.CFrame:Inverse() * LeftArm.CFrame) end
	if RightArm and ProxyRightArm then ProxyRightArm.CFrame = Hrp0.CFrame * (Hrp1.CFrame:Inverse() * RightArm.CFrame) end

	if RightLeg and ProxyRightLeg then ProxyRightLeg.CFrame = Hrp0.CFrame * (Hrp1.CFrame:Inverse() * RightLeg.CFrame) end

	if finished then return end

	local currentTime = tick()
	local elapsedTime = currentTime - startTime
	local index, alpha, isFinished = calculateMainTimeline(elapsedTime)

	if isFinished or index >= totalFrames then
		finished = true

		Hrp0.CFrame = Hrp1.CFrame
		Hrp0.Velocity = Hrp1.Velocity

local PriorityLayers = {
    Enum.AnimationPriority.Core,
    Enum.AnimationPriority.Idle,
    Enum.AnimationPriority.Movement,
    Enum.AnimationPriority.Action,
    Enum.AnimationPriority.Action2,
    Enum.AnimationPriority.Action3,
    Enum.AnimationPriority.Action4
}

local function SetLayer(track, layer)
    layer = math.clamp(layer, 1, #PriorityLayers)
    track.Priority = PriorityLayers[layer]
end

local function playAnim(id, layer, fadeTime, speed, timePos, stopDelay)
    local animation = Instance.new("Animation")
    animation.AnimationId = "rbxassetid://" .. tostring(id)

    local track = humanoid:LoadAnimation(animation)
    SetLayer(track, layer)

    track:Play(fadeTime)
    track.TimePosition = timePos or 0
    track:AdjustSpeed(speed)

    if stopDelay then
        task.delay(stopDelay, function()
            track:Stop(fadeTime)
        end)
    end

    return track
end
