local targetFPS = 55        
local playbackSpeed = 0.67

local totalFrames = #frames
local totalDuration = ((totalFrames - 1) / targetFPS) / playbackSpeed
local origin = frames[1]:Inverse()

local function calculateMainTimeline(elapsedTime)
	if elapsedTime >= totalDuration then
		return totalFrames, 1, true
	end
	
	local progress = elapsedTime / totalDuration
	local scaledIndex = progress * (totalFrames - 1) + 1
	local index = math.floor(scaledIndex)
	local alpha = scaledIndex - index
	
	return index, alpha, false
end

local currentTrack = nil
local function PlayAnimation(animId)
	local anim = Instance.new("Animation")
	anim.AnimationId = typeof(animId) == "number" and ("rbxassetid://" .. tostring(animId)) or animId
	currentTrack = Animator:LoadAnimation(anim)
	currentTrack.Priority = Enum.AnimationPriority.Action4
	currentTrack:Play()
	return currentTrack
end

	if RightLeg and ProxyRightLeg then ProxyRightLeg.CFrame = Hrp0.CFrame * Hrp1.CFrame:ToObjectSpace(RightLeg.CFrame) end

	if finished then return end

	local currentTime = tick()
	local elapsedTime = currentTime - startTime
	local index, alpha, isFinished = calculateMainTimeline(elapsedTime)

	if isFinished or index >= totalFrames then
		finished = true

		if currentTrack then
			currentTrack:Stop()
		end

		Hrp0.CFrame = Hrp1.CFrame
		Hrp0.Velocity = Hrp1.Velocity

		if Hrp0:FindFirstChild("RootJoint") then
			Hrp0.RootJoint.Part0 = Hrp0
		end
		Hrp0.Parent = Character
		Hrp1:Destroy()

		if Proxy then Proxy:Destroy() end
		if invisibleBlock then invisibleBlock:Destroy() end

		DisconnectAll()
		return
	end

	local nextIndex = math.min(index + 1, totalFrames)
	local offsetA = origin * frames[index]
	local offsetB = origin * frames[nextIndex]
	
	local targetA = Hrp1.CFrame * offsetA
	local targetB = Hrp1.CFrame * offsetB

	Hrp0.CFrame = targetA:Lerp(targetB, alpha)
	Hrp0.Velocity = Hrp1.Velocity

local PriorityLayers = {
    Enum.AnimationPriority.Core,
    Enum.AnimationPriority.Idle,
    Enum.AnimationPriority.Movement,
    Enum.AnimationPriority.Action,
    Enum.AnimationPriority.Action2,
    Enum.AnimationPriority.Action3,
    Enum.AnimationPriority.Action4
}

local function SetLayer(track, layer)
    layer = math.clamp(layer, 1, #PriorityLayers)
    track.Priority = PriorityLayers[layer]
end

local function playAnim(id, layer, fadeTime, speed, stopDelay)
    local animation = Instance.new("Animation")
    animation.AnimationId = "rbxassetid://" .. tostring(id)

    local track = humanoid:LoadAnimation(animation)
    SetLayer(track, layer)

    track:Play(fadeTime)
    track.TimePosition = 0
    track:AdjustSpeed(speed)

    if stopDelay then
        task.delay(stopDelay, function()
            track:Stop(fadeTime)
        end)
    end

    return track
end
